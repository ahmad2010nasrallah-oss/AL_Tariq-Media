function goToPage(pageId){ /* ========================================== تغيير الصفحة النشطة ========================================== */ $$(".page").forEach(page => { page.classList.toggle( "active", page.id === pageId ); }); /* ========================================== تغيير زر القائمة الجانبية النشط ========================================== */ $$(".nav-btn").forEach(button => { button.classList.toggle( "active", button.dataset.page === pageId ); }); /* ========================================== عنوان الصفحة ========================================== */ const titles = { dashboard: "Dashboard", works: "Our Works", services: "Services", media: "Media Library", settings: "Settings" }; if (quickAddButton) { if (pageId === "works") { quickAddButton.style.display = "inline-flex"; } else { quickAddButton.style.display = "none"; } } /* ========================================== إغلاق القائمة الجانبية في الموبايل ========================================== */ $("#sidebar") ?.classList .remove("open"); /* ========================================== الرجوع لأعلى الصفحة ========================================== */ window.scrollTo({ top: 0, behavior: "smooth" }); } /* ========================================================= أزرار التنقل ========================================================= */ $$(".nav-btn").forEach(button => { button.addEventListener( "click", () => { goToPage( button.dataset.page ); } ); }); /* ========================================================= أزرار data-go ========================================================= */ $$("[data-go]").forEach(button => { button.addEventListener( "click", () => { goToPage( button.dataset.go ); } ); }); /* ========================================================= MOBILE MENU ========================================================= */ $("#mobileMenu") ?.addEventListener( "click", () => { $("#sidebar") ?.classList .toggle("open"); } ); /* ========================================================= زر إضافة مشروع الموجود في الهيدر يعمل فقط إذا كنا في صفحة works ========================================================= */ $("#quickAdd") ?.addEventListener( "click", () => { const worksPage = $("#works");
